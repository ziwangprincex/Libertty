# Libertty's command specs, for autocomplete's list (⌃⇧Space): subcommands and options of
# common tools, with what each does. One entry a line, indented two spaces a level:
#
#   NAME[|ALIAS…]  DESCRIPTION     a subcommand (or the command, unindented)
#   -X|--long[=]  DESCRIPTION      an option; one ending in = takes a value
#   @KIND                          what its arguments are: path, dir, branch, remote,
#                                  host (~/.ssh/config), script (package.json), target (Makefile)
#
# A description is everything after two or more spaces.

cd  Change the shell's folder
  @dir
ls  List a folder
  -l  Long listing
  -a  Include hidden files
  -la  Long listing, hidden files too
  -h  Human-readable sizes
  @path
cat  Print files
  @path

  @path

  -a=  Open with this app
  -R  Reveal in Finder
  @path

  @path

  @path
rm  Remove files
  -r  Folders and what's in them
  -f  Don't ask
  -rf  Folders, without asking
  @path

  -p  With its parents
  @dir

  -r  Folders too
  @path
mv  Move or rename files
  @path

  -p=  Port

  -r  Folders too
  @path
git  Distributed version control
  status|st  Show the working tree's state
    -s|--short  Short format
    -b|--branch  Show the branch
  add  Stage changes
    -A|--all  Every change, new files too
    -p|--patch  Pick hunks
    -u|--update  Tracked files only
    @path
  commit|ci  Record staged changes
    -m|--message=  The message
    -a|--all  Stage tracked changes first
    --amend  Redo the last commit
    --no-edit  Keep the message
    --fixup=  A fixup for a commit
    -S|--gpg-sign  Sign it
  push  Send commits to a remote
    -u|--set-upstream  Track the branch there
    -f|--force  Overwrite the remote's
    --force-with-lease  Overwrite only what you've seen
    --tags  Tags too
    @remote
  pull  Fetch and merge
    --rebase  Rebase instead of merging
    --ff-only  Only fast-forward
    @remote
  fetch  Download from a remote
    --all  Every remote
    -p|--prune  Drop what's gone there
    @remote
  checkout|co  Switch branches or restore files
    -b=  A new branch
    @branch
  switch|sw  Switch branches
    -c|--create=  A new branch
    @branch
  branch|br  List, make or delete branches
    -d|--delete=  Delete a merged branch
    -D=  Delete a branch
    -m|--move=  Rename a branch
    -a|--all  Remote branches too
    -v|--verbose  With their last commits
    @branch
  merge  Join a branch into this one
    --no-ff  Always a merge commit
    --squash  One change, no merge
    --abort  Give up a conflicted merge
    @branch
  rebase  Reapply commits on another base
    -i|--interactive  Edit the list first
    --continue  After resolving
    --abort  Give it up
    --onto=  Onto this base
    @branch
  log  Show commits
    --oneline  One line each
    --graph  Draw the history
    -p|--patch  With their changes
    -n=  Only this many
    --author=  By this author
    --stat  Files changed
  diff  Show changes
    --staged|--cached  What's staged
    --stat  Files changed
    --name-only  Only the files' names
    @path
  show  Show a commit
    --stat  Files changed
  stash  Put changes aside
    push  Put them aside
    pop  Take the last back
    apply  Apply one, keep it
    list  List them
    drop  Delete one
    show  Show one
  restore  Restore files
    --staged  Unstage
    --source=  From this commit
    @path
  reset  Move HEAD
    --soft  Keep changes staged
    --hard  Discard changes
    --mixed  Keep changes unstaged
  clone  Copy a repository
    --depth=  Only the latest commits
    --recurse-submodules  With submodules
  remote  Manage remotes
    add  Add one
    remove  Remove one
    -v  With their URLs
  tag  Create or list tags
    -a  Annotated
    -d  Delete
  worktree  Manage worktrees
    add  Add one
    list  List them
    remove  Remove one
  cherry-pick  Apply a commit here
    --continue  After resolving
    --abort  Give it up
  revert  Undo a commit with a new one
  blame  Who changed each line
    @path
  grep  Search tracked files
  bisect  Find a commit by halving
    start  Begin
    good  This one is good
    bad  This one is bad
    reset  End
  submodule  Manage submodules
    update  Update them
      --init  Set up new ones
      --recursive  Theirs too
  init  Make a repository
  config  Settings
    --global  Yours
    --local  This repository's
